    <script>
        class StaticMCPChatbot {
            constructor() {
                this.messages = [];
                this.mcpContent = null;
                this.isLoading = false;
                this.initializeEventListeners();
                this.updateStatus(false);
            }

            initializeEventListeners() {
                const input = document.getElementById('chatInput');
                const sendButton = document.getElementById('sendButton');

                input.addEventListener('keydown', (e) => {
                    if (e.key === 'Enter' && !e.shiftKey) {
                        e.preventDefault();
                        this.sendMessage();
                    }
                });

                input.addEventListener('input', () => {
                    this.autoResize(input);
                });
            }

            autoResize(textarea) {
                textarea.style.height = 'auto';
                textarea.style.height = Math.min(textarea.scrollHeight, 120) + 'px';
            }

            updateStatus(connected) {
                const indicator = document.getElementById('statusIndicator');
                indicator.classList.toggle('connected', connected);
            }

            async loadMCPContent() {
                const mcpUrl = document.getElementById('mcpUrl').value;
                if (!mcpUrl) {
                    this.addMessage('error', 'Please enter a valid MCP content URL');
                    return;
                }

                try {
                    this.addMessage('system', 'Loading MCP content...');
                    const response = await fetch(mcpUrl);

                    if (!response.ok) {
                        throw new Error(`Failed to fetch MCP content: ${response.status}`);
                    }

                    this.mcpContent = await response.json();
                    this.addMessage('system', `MCP content loaded successfully! Found ${Object.keys(this.mcpContent).length} resources`);
                    this.updateStatus(true);
                } catch (error) {
                    console.error('Error loading MCP content:', error);
                    this.addMessage('error', `Failed to load MCP content: ${error.message}`);
                    this.updateStatus(false);
                }
            }

            async sendMessage() {
                const input = document.getElementById('chatInput');
                const message = input.value.trim();

                if (!message || this.isLoading) return;

               /* const apiKey = document.getElementById('apiKey').value;
                if (!apiKey) {
                    this.addMessage('error', 'Please enter your API key');
                    return;
                }*/

                input.value = '';
                input.style.height = 'auto';
                this.addMessage('user', message);

                this.isLoading = true;
                this.showTyping();

                try {
                    const response = await this.callLLM(message, apiKey);
                    this.hideTyping();
                    this.addMessage('assistant', response);
                } catch (error) {
                    this.hideTyping();
                    console.error('Error calling LLM:', error);
                    this.addMessage('error', `Error: ${error.message}`);
                } finally {
                    this.isLoading = false;
                }
            }

            async callLLM(message, apiKey) {
                const provider = document.getElementById('apiProvider').value;

                // Prepare context with MCP content if available
                let systemMessage = "You are a helpful AI assistant.";
                let contextMessage = message;

                if (this.mcpContent) {
                    systemMessage += " You have access to additional context data that may be relevant to user queries.";

                    // Simple keyword matching to find relevant MCP content
                    const relevantContent = this.findRelevantMCPContent(message);
                    if (relevantContent.length > 0) {
                        contextMessage = `${message}\n\nRelevant context:\n${relevantContent.map(item =>
                            `${item.title}: ${item.content}`
                        ).join('\n\n')}`;
                    }
                }

                const messages = [
                    { role: "system", content: systemMessage },
                    ...this.messages.slice(-10), // Keep last 10 messages for context
                    { role: "user", content: contextMessage }
                ];

                switch (provider) {
                    case 'openai':
                        return await this.callOpenAI(messages, apiKey);
                    case 'anthropic':
                        return await this.callAnthropic(messages, apiKey);
                    case 'cohere':
                        return await this.callCohere(messages, apiKey);
                    case 'openrouter':
                        return await this.callOpenRouter(messages, apiKey);
                    default:
                        throw new Error('Unsupported API provider');
                }
            }

            findRelevantMCPContent(query) {
                if (!this.mcpContent) return [];

                const keywords = query.toLowerCase().split(' ').filter(word => word.length > 3);
                const relevant = [];

                for (const [key, content] of Object.entries(this.mcpContent)) {
                    const contentText = JSON.stringify(content).toLowerCase();
                    const relevanceScore = keywords.reduce((score, keyword) => {
                        return score + (contentText.includes(keyword) ? 1 : 0);
                    }, 0);

                    if (relevanceScore > 0) {
                        relevant.push({
                            title: key,
                            content: typeof content === 'string' ? content : JSON.stringify(content, null, 2),
                            score: relevanceScore
                        });
                    }
                }

                return relevant.sort((a, b) => b.score - a.score).slice(0, 3);
            }

            async callOpenAI(messages, apiKey) {
                const response = await fetch('https://api.openai.com/v1/chat/completions', {
                    method: 'POST',
                    headers: {
                        'Authorization': `Bearer ${apiKey}`,
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({
                        model: 'gpt-3.5-turbo',
                        messages: messages,
                        max_tokens: 1000,
                        temperature: 0.7
                    })
                });

                if (!response.ok) {
                    const error = await response.json();
                    throw new Error(error.error?.message || 'OpenAI API error');
                }

                const data = await response.json();
                return data.choices[0].message.content;
            }

            async callAnthropic(messages, apiKey) {
                const systemMessage = messages.find(m => m.role === 'system')?.content || '';
                const chatMessages = messages.filter(m => m.role !== 'system');

                const response = await fetch('https://api.anthropic.com/v1/messages', {
                    method: 'POST',
                    headers: {
                        'x-api-key': apiKey,
                        'Content-Type': 'application/json',
                        'anthropic-version': '2023-06-01'
                    },
                    body: JSON.stringify({
                        model: 'claude-3-sonnet-20240229',
                        max_tokens: 1000,
                        system: systemMessage,
                        messages: chatMessages
                    })
                });

                if (!response.ok) {
                    const error = await response.json();
                    throw new Error(error.error?.message || 'Anthropic API error');
                }

                const data = await response.json();
                return data.content[0].text;
            }

            async callCohere(messages, apiKey) {
                const lastMessage = messages[messages.length - 1].content;
                const chatHistory = messages.slice(0, -1).map(m => ({
                    role: m.role === 'assistant' ? 'CHATBOT' : 'USER',
                    message: m.content
                }));

                const response = await fetch('https://api.cohere.ai/v1/chat', {
                    method: 'POST',
                    headers: {
                        'Authorization': `Bearer ${apiKey}`,
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({
                        message: lastMessage,
                        chat_history: chatHistory,
                        model: 'command',
                        max_tokens: 1000
                    })
                });

                if (!response.ok) {
                    const error = await response.json();
                    throw new Error(error.message || 'Cohere API error');
                }

                const data = await response.json();
                return data.text;
            }

            async callOpenRouter(messages, apiKey) {
                const response = await fetch('https://openrouter.ai/api/v1/chat/completions', {
                    method: 'POST',
                    headers: {
                        'Authorization': `Bearer ${apiKey}`,
                        'Content-Type': 'application/json',
                        'HTTP-Referer': window.location.origin,
                        'X-Title': 'StaticMCP Chatbot'
                    },
                    body: JSON.stringify({
                        model: 'meta-llama/llama-2-7b-chat',
                        messages: messages,
                        max_tokens: 1000,
                        temperature: 0.7
                    })
                });

                if (!response.ok) {
                    const error = await response.json();
                    throw new Error(error.error?.message || 'OpenRouter API error');
                }

                const data = await response.json();
                return data.choices[0].message.content;
            }

            addMessage(type, content) {
                const messagesContainer = document.getElementById('chatMessages');
                const messageDiv = document.createElement('div');
                messageDiv.className = `message ${type}`;
                messageDiv.textContent = content;

                // Add to conversation history (except system/error messages)
                if (type === 'user' || type === 'assistant') {
                    this.messages.push({
                        role: type === 'user' ? 'user' : 'assistant',
                        content: content
                    });
                }

                // Add MCP content indicator for assistant messages
                if (type === 'assistant' && this.mcpContent) {
                    const relevantContent = this.findRelevantMCPContent(content);
                    if (relevantContent.length > 0) {
                        const mcpDiv = document.createElement('div');
                        mcpDiv.className = 'mcp-content';
                        mcpDiv.innerHTML = `<h4>?? Used MCP Resources:</h4>${relevantContent.map(item => item.title).join(', ')}`;
                        messageDiv.appendChild(mcpDiv);
                    }
                }

                messagesContainer.appendChild(messageDiv);
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            showTyping() {
                const messagesContainer = document.getElementById('chatMessages');
                const typingDiv = document.createElement('div');
                typingDiv.className = 'typing-indicator';
                typingDiv.id = 'typingIndicator';
                typingDiv.innerHTML = `
                    <div class="typing-dots">
                        <span></span>
                        <span></span>
                        <span></span>
                    </div>
                `;
                messagesContainer.appendChild(typingDiv);
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            hideTyping() {
                const typingDiv = document.getElementById('typingIndicator');
                if (typingDiv) {
                    typingDiv.remove();
                }
            }
        }

        // Initialize the chatbot
        const chatbot = new StaticMCPChatbot();

        // Global functions for button clicks
        function sendMessage() {
            chatbot.sendMessage();
        }

        function loadMCPContent() {
            chatbot.loadMCPContent();
        }
    </script>
